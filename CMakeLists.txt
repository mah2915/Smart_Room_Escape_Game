cmake_minimum_required(VERSION 3.16)
project(SmartEscapeRoom VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(escape_engine src/Game.cpp)
target_include_directories(escape_engine PUBLIC include)
if(MSVC)
 target_compile_options(escape_engine PRIVATE /W4)
else()
 target_compile_options(escape_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(smart_escape_room src/main.cpp)
target_link_libraries(smart_escape_room PRIVATE escape_engine)
include(CTest)
if(BUILD_TESTING)
 add_executable(game_tests tests/test_game.cpp)
 target_link_libraries(game_tests PRIVATE escape_engine)
 add_test(NAME game_regression COMMAND game_tests)
endif()
